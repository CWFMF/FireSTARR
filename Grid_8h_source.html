340 string saveToAsciiFile(
341 const string_view dir,
342 const string_view base_name,
343 std::function<R(T value)> convert,
344 const R no_data
345 ) const
346 {
347 const auto [min_x, min_y, max_x, max_y] = dataBounds();
348 // auto bounds = dataBounds();
349 // auto min_x = std::get<0>(bounds);
350 // auto min_y = std::get<1>(bounds);
351 // auto max_x = std::get<2>(bounds);
352 // auto max_y = std::get<3>(bounds);
353#ifdef DEBUG_GRIDS
354 logging::debug("Bounds are ({:d}, {:d}), ({:d}, {:d})", min_x, min_y, max_x, max_y);
355#endif
356 logging::extensive("Lower left corner is ({:d}, {:d})", min_x, min_y);
357 logging::extensive("Upper right corner is ({:d}, {:d})", max_x, max_y);
358 const MathSize xll = this->xllcorner() + min_x * this->cellSize();
359 // offset is different for y since it's flipped
360 const MathSize yll = this->yllcorner() + (min_y) * this->cellSize();
361 logging::extensive("Lower left corner is ({:f}, {:f})", xll, yll);
362 // HACK: make sure it's always at least 1
363 const auto height_calc = static_cast<MathSize>(max_y) - min_y + 1;
364 const auto width_calc = static_cast<MathSize>(max_x) - min_x + 1;
365 const auto filename = create_file_name(dir, base_name, "asc");
366 ofstream out{filename};
367 write_ascii_header(
368 out, width_calc, height_calc, xll, yll, this->cellSize(), static_cast<MathSize>(no_data)
369 );
370 for (Idx y0 = 0; y0 < height_calc; ++y0)
371 {
372 // HACK: do this so that we always get at least one pixel in output
373 // need to output in reverse order since (0,0) is bottom left
374 const Idx y1 = static_cast<Idx>(max_y) - y0;
375 for (Idx x0 = 0; x0 < width_calc; ++x0)
376 {
377 const XYIdx idx{static_cast<Idx>(min_x + x0), static_cast<Idx>(y1)};
378 // HACK: use + here so that it gets promoted to a printable number
379 // prevents char type being output as characters
380 out << +(convert(this->at(idx))) << " ";
381 }
382 out << "\n";
383 }
384 out.close();
385 this->createPrj(dir, base_name);
386 return filename;
387 }
