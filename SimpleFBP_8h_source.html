26[[nodiscard]] constexpr int calculate_grass_curing(const int nd)
27{
28 const auto curing = (nd < START_GREENING)
29 ? // we're before foliar moisture dip has started
30 100
31 : (nd >= 50)
32 ? 0 // foliar moisture is at 120.0, so grass should be totally uncured
33 // HACK: invent a formula that has 50% curing at the bottom of the foliar
34 // moisture dip foliar moisture above ranges between 120 and 85, with 85
35 // being at the point where we want 50% cured Curing:
36 // -43 => 100, 0 => 50, 50 => 0 least-squares best fit:
37 : static_cast<int>(52.5042 - 1.07324 * nd);
38 return max(0, min(100, curing));
39}
