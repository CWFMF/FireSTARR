100 [[nodiscard]] static ConstantGrid<T, V> readTiff(
101 const string_view filename,
102 const Point& point,
103 std::function<T(V, V)> convert
104 )
105 {
106 logging::debug("Reading a raster where T = {:s}, V = {:s}", typeid(T).name(), typeid(V).name());
107 auto grid = readTiffInt(filename, point);
108 const auto nodata_input = grid.nodataInput();
109 T nodata_value = convert(nodata_input, nodata_input);
110 logging::check_fatal(
111 convert(nodata_input, nodata_input) != nodata_value,
112 "Expected nodata value to be returned from convert()"
113 );
114 vector<T> values(static_cast<size_t>(MAX_HEIGHT) * MAX_WIDTH, nodata_value);
115 std::transform(grid.data.begin(), grid.data.end(), values.begin(), [&](const int v) {
116 return convert(static_cast<V>(v), nodata_input);
117 });
118 ConstantGrid<T, V> result{
119 grid.cellSize(),
120 grid.width(),
121 grid.height(),
122 static_cast<V>(grid.nodataInput()),
123 nodata_value,
124 grid.xllcorner(),
125 grid.yllcorner(),
126 grid.xurcorner(),
127 grid.yurcorner(),
128 grid.proj4(),
129 std::move(values)
130 };
131 return result;
132 }
