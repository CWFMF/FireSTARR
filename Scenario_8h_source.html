297 [[nodiscard]] bool survives(
298 const DurationSize time,
299 const Cell& cell,
300 const DurationSize time_at_location
301 ) const
302 {
303 // HACK: resolve once and fail if not set already
304 static const auto& settings = fs::settings::instance();
305 if (settings.deterministic)
306 {
307 // always survive if deterministic
308 return true;
309 }
310 try
311 {
312 const auto fire_wx = weather_;
313 const auto wx = fire_wx->at(time);
314 // use Mike's table
315 // days at location equivalent DMC
316 // * 54.40794725611335
317 // 5 49.77788646253319
318 // 4 45.15330890865478
319 // 3 40.184467357005346
320 // 2 35.025698388961054
321 // 1 15.049926856936347
322 const auto mc = wx->mcDmcPct();
323 if (100 > mc || (109 >= mc && 5 > time_at_location) || (119 >= mc && 4 > time_at_location)
324 || (131 >= mc && 3 > time_at_location) || (145 >= mc && 2 > time_at_location)
325 || (218 >= mc && 1 > time_at_location))
326 {
327 return true;
328 }
329 // we can look by fuel type because the entire landscape shares the weather
330 return extinctionThreshold(time) < fire_wx->survivalProbability(time, cell.fuelCode());
331 }
332 catch (const std::out_of_range&)
333 {
334 // no weather, so don't survive
335 return false;
336 }
337 }
