81 [[nodiscard]] ThresholdSize survivalProbability(const FwiWeather& wx) const noexcept override
82 {
83 // divide by 100 since we need moisture ratio
84 // IFERROR(((1 / (1 + EXP($G$43 + $I$43 *
85 // (Q$44 * $O$43 + $N$43)))) -
86 // (1 / (1 + EXP($G$43 + $I$43 * (2.5 * $O$43 + $N$43)))))
87 // / (1 / (1 + EXP($G$43 + $I$43 * $N$43))), 0)
88 // HACK: use same constants for all fuels because they seem to work nicer than
89 // using the ratios, but they change anyway because of the other fuel attributes
90 static const auto WFfmc = 0.25;
91 static const auto WDmc = 1.0;
92 static const auto RatioHartford = 0.5;
93 static const auto RatioFrandsen = 1.0 - RatioHartford;
94 static const auto RatioAspen = 0.5;
95 static const auto RatioFuel = 1.0 - RatioAspen;
96 const auto mc_ffmc = wx.mcFfmc() * WFfmc + WDmc;
97 static const auto McFfmcSaturated = 2.5 * WFfmc + WDmc;
98 static const auto McDmc = WDmc;
99 const auto prob_ffmc_peat = probabilityPeat(mc_ffmc);
100 const auto prob_ffmc_peat_saturated = probabilityPeat(McFfmcSaturated);
101 const auto prob_ffmc_peat_zero = probabilityPeat(McDmc);
102 const auto prob_ffmc_peat_weighted =
103 (prob_ffmc_peat - prob_ffmc_peat_saturated) / prob_ffmc_peat_zero;
104 const auto prob_ffmc = duffFfmcType()->probabilityOfSurvival(mc_ffmc * 100);
105 const auto prob_ffmc_saturated = duffFfmcType()->probabilityOfSurvival(McFfmcSaturated * 100);
106 const auto prob_ffmc_zero = duffFfmcType()->probabilityOfSurvival(McDmc);
107 const auto prob_ffmc_weighted = (prob_ffmc - prob_ffmc_saturated) / prob_ffmc_zero;
108 const auto term_otway = exp(-3.11 + 0.12 * wx.dmc.value);
109 const auto prob_otway = term_otway / (1 + term_otway);
110 const auto mc_pct = wx.mcDmcPct() * dmcRatio() + wx.mcFfmcPct() * ffmcRatio();
111 const auto prob_weight_ffmc = duffFfmcType()->probabilityOfSurvival(mc_pct);
112 const auto prob_weight_ffmc_peat = probabilityPeat(mc_pct / 100);
113 const auto prob_weight_dmc = duffDmcType()->probabilityOfSurvival(wx.mcDmcPct());
114 const auto prob_weight_dmc_peat = probabilityPeat(wx.mcDmc());
115 // chance of survival is 1 - chance of it not surviving in every fuel
116 const auto tot_prob =
117 1
118 - (1 - prob_ffmc_peat_weighted) * (1 - prob_ffmc_weighted)
119 * ((1 - prob_otway) * RatioAspen + ((1 - prob_weight_ffmc_peat) * RatioHartford + (1 - prob_weight_ffmc) * RatioFrandsen) * ((1 - prob_weight_dmc_peat) * RatioHartford + (1 - prob_weight_dmc) * RatioFrandsen) * RatioFuel);
120 return tot_prob;
121 }
