25[[nodiscard]] constexpr int calculate_grass_curing(const int nd)
26{
27 const auto curing = (nd < START_GREENING)
28 ? // we're before foliar moisture dip has started
29 100
30 : (nd >= 50)
31 ? 0 // foliar moisture is at 120.0, so grass should be totally uncured
32 // HACK: invent a formula that has 50% curing at the bottom of the foliar
33 // moisture dip foliar moisture above ranges between 120 and 85, with 85
34 // being at the point where we want 50% cured Curing:
35 // -43 => 100, 0 => 50, 50 => 0 least-squares best fit:
36 : static_cast<int>(52.5042 - 1.07324 * nd);
37 return max(0, min(100, curing));
38}
