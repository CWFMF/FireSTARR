149 tuple<Idx, Idx, Idx, Idx> dataBounds() const override
150 {
151 Idx min_y = this->height();
152 Idx max_y = 0;
153 Idx min_x = this->width();
154 Idx max_x = 0;
155 for (const auto& kv : this->data)
156 {
157 const Idx r = kv.first.y_value();
158 const Idx c = kv.first.x_value();
159 min_y = min(min_y, r);
160 max_y = max(max_y, r);
161 min_x = min(min_x, c);
162 max_x = max(max_x, c);
163 }
164 // do this so that we take the center point when there's no data since it should
165 // stay the same if the grid is centered on the fire
166 if (min_y > max_y)
167 {
168 min_y = max_y = this->height() / 2;
169 }
170 if (min_x > max_x)
171 {
172 min_x = max_x = this->width() / 2;
173 }

210 [[nodiscard]] list<XYIdx> makeEdge() const
211 {
212 list<XYIdx> edge{};
213 for (const auto& kv : this->data)
214 {
215 auto loc = kv.first;
216 // FIX: any way to avoid decomposing?
217 auto [x_loc, y_loc] = hash_to_xy(loc);
218 auto on_edge = false;
219 for (Idx r = -1; !on_edge && r <= 1; ++r)
220 {
221 const Idx y = y_loc.value + r;
222 if (!(y < 0 || y >= this->height()))
223 {
224 for (Idx c = -1; !on_edge && c <= 1; ++c)
225 {
226 const Idx x = x_loc.value + c;
227 if (!(x < 0 || x >= this->width()) && this->data.find(XYIdx{x, y}) == this->data.end())
228 {
229 on_edge = true;
230 }
231 }
232 }
233 }
234 if (on_edge)
235 {
236 edge.push_back(loc);
237 }
238 }
239 logging::info(
240 "Created edge for perimeter with length {:d} m",
241 static_cast<size_t>(this->cellSize() * edge.size())
242 );
243 return edge;
244 }
