175template <class TypeA, class TypeB>
176int compare_duff(
177 const string name,
178 const TypeA& a,
179 const TypeB& b,
180 const logging::level log_level = logging::level::info
181)
182{
183 static constexpr int RESOLUTION = 10000;
184 static constexpr MathSize RANGE = 250.0;
185 // check %, so 1 decimal is fine
186 static constexpr auto EPSILON = static_cast<MathSize>(1e-1);
187 std::ignore = logging::output(log_level, "Checking {:s}", name);
188 logging::check_equal_verbose(logging::level::debug, a.ash, b.ash, "ash");
189 logging::check_equal_verbose(logging::level::debug, a.rho, b.rho, "rho");
190 logging::check_equal_verbose(logging::level::debug, a.b0, b.b0, "b0");
191 logging::check_equal_verbose(logging::level::debug, a.b1, b.b1, "b1");
192 logging::check_equal_verbose(logging::level::debug, a.b2, b.b2, "b2");
193 logging::check_equal_verbose(logging::level::debug, a.b3, b.b3, "b3");
194 for (auto i = 0; i < RESOLUTION; ++i)
195 {
196 const MathSize mc = RANGE * i / RESOLUTION;
197 const auto msg = std::format("probability of survival (mc = {})", mc);
198 logging::check_tolerance(
199 EPSILON, a.probabilityOfSurvival(mc), b.probabilityOfSurvival(mc), msg.c_str()
200 );
201 }
202 return 0;
203}
